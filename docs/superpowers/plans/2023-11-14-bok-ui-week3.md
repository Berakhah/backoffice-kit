# bok-ui v0.1 — Implementation Plan (Week 3)

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship `bok-ui` v0.1.0 to npm — the token layer whole, nine components, and a Storybook on GitHub Pages with axe passing — before `agent-perimeter`'s week 4 needs it.

**Architecture:** Tokens land whole; components land lazily. `bok-ui` publishes to npm, so adding a component later is a minor version bump while changing a token later is a breaking visual change across four repos. That asymmetry decides the split: the OKLCH ramp, the semantic scales, the density *mechanism*, the a11y contract and the print sheet are complete in v0.1; four components and all chart primitives defer to the phase that first demands them.

**Tech Stack:** TypeScript (strict), React 19, Vite (library mode), Vitest + Testing Library, Storybook 8, `@storybook/addon-a11y` + `axe-core`, npm trusted publishing (OIDC).

**Spec:** `docs/superpowers/specs/2023-11-14-backoffice-kit-design.md` §8
**Interface contract:** `docs/INTERFACE-CONTRACT.md` §6 — the component surface the four consumer repos check against.
**Companion plan:** `2023-11-14-bok-core-week0.md`. `RailPayload` is produced by `bok_core.provenance.render.to_rail()`; this package consumes that exact JSON shape.

## Global Constraints

- **TypeScript strict.** No `any` without a comment justifying it on the same line.
- **Light mode is primary.** Dark exists and is excellent; light is what you design first and what screenshots use.
- **Severity and provenance state are never encoded in colour alone.** Every semantic colour ships a glyph *and* a text label. Two target markets are regulated for accessibility.
- **Tabular numerals wherever a number appears.** A column of figures that shifts as it updates is the fastest way to look amateur to a controller.
- **Self-hosted fonts only.** No Google Fonts CDN call — a privacy and offline-demo liability.
- **`prefers-reduced-motion` disables all motion.** Not reduces — disables.
- **axe-core: zero serious or critical violations**, enforced in CI.
- **Reject three looks explicitly:** warm-cream + serif + terracotta; near-black + single acid accent; broadsheet with hairlines and zero radius. If a screen could drop into any of the four products unchanged and still look right, it is too generic.
- Every exported symbol must appear in `docs/INTERFACE-CONTRACT.md` §6.

```ts
// Severity and provenance state are NEVER encoded in colour alone. Two of the four
// target markets are literally regulated for accessibility; a colour-only encoding
// would be an own goal in a sales conversation.

export interface SemanticEntry {
  readonly token: `--bok-${string}`;
  readonly glyph: string;
  readonly label: string;
}

export const SEVERITY = {
  CRITICAL: { token: "--bok-critical", glyph: "◆", label: "Critical" },
  HIGH: { token: "--bok-high", glyph: "▲", label: "High" },
  MEDIUM: { token: "--bok-medium", glyph: "■", label: "Medium" },
  LOW: { token: "--bok-low", glyph: "●", label: "Low" },
  INFO: { token: "--bok-info", glyph: "○", label: "Info" },
} as const satisfies Record<string, SemanticEntry>;

export const PROVENANCE_STATE = {
  verified: { token: "--bok-verified", glyph: "✓", label: "Verified" },
  modelled: { token: "--bok-modelled", glyph: "~", label: "Modelled" },
  unverified: { token: "--bok-unverified", glyph: "?", label: "Unverified" },
} as const satisfies Record<string, SemanticEntry>;

export type SeverityKey = keyof typeof SEVERITY;
export type ProvenanceStateKey = keyof typeof PROVENANCE_STATE;
```

```tsx
import { useEffect } from "react";
import { PROVENANCE_STATE } from "./semantics";
import { provenanceStateOf, type RailNode, type RailPayload } from "./types";

export interface ProvenanceRailProps {
  payload: RailPayload;
  open: boolean;
  onClose: () => void;
}

function sourceLine(node: RailNode): string {
  const s = node.source as Record<string, unknown>;
  switch (node.source_kind) {
    case "file":
      return `${String(s.path)}:${String(s.line)}`;
    case "document":
      return `${String(s.path)} · page ${String(s.page)}`;
    case "url":
      return `${String(s.url)} · retrieved ${String(s.retrieved_at).slice(0, 10)}`;
    case "model":
      return `model call ${String(s.ledger_call_id)}`;
    case "human":
      return `${String(s.who)} · ${String(s.at).slice(0, 10)}`;
    case "rule":
      return `${String(s.authority)} ${String(s.citation)} · confirmed by ${String(s.confirmed_by)}`;
    case "dom":
      return `${String(s.selector)} · snapshot ${String(s.snapshot_id)} · rung ${String(s.rung)}`;
    default:
      return node.source_kind;
  }
}

function Node({ node, index }: { node: RailNode; index: number }) {
  const state = provenanceStateOf(node);
  const entry = PROVENANCE_STATE[state];
  return (
    <li
      style={{
        // Stagger is disabled wholesale by the reduced-motion block in theme.css.
        animation: `bok-rail-in 180ms ease-out ${index * 30}ms both`,
        borderLeft: `2px solid var(${entry.token})`,
        paddingLeft: "var(--bok-density-pad-x)",
        marginBottom: "calc(var(--bok-space) * 3)",
      }}
    >
      <div data-bok-numeric style={{ fontSize: "1.125rem" }}>{String(node.value)}</div>
      <div data-testid="provenance-marker" data-glyph={entry.glyph}>
        {entry.glyph} {entry.label} · {node.method}
        {node.derivation ? ` (${node.derivation})` : ""}
      </div>
      <div style={{ fontFamily: "var(--bok-font-mono)", fontSize: "0.8125rem" }}>
        {sourceLine(node)}
      </div>
      {node.confidence !== null && (
        <div>
          confidence {node.confidence.toFixed(2)}
          {node.calibration_state === "UNCALIBRATED" && (
            <> · <strong>uncalibrated</strong> — {node.calibration_basis}</>
          )}
        </div>
      )}
      <div>{String(node.observed_at).slice(0, 19).replace("T", " ")}</div>
      {node.caveat && <div style={{ fontStyle: "italic" }}>{node.caveat}</div>}
      {node.parents.length > 0 && (
        <ol style={{ listStyle: "none", paddingLeft: "calc(var(--bok-space) * 3)" }}>
          {node.parents.map((p, i) => (
            <Node key={`${p.source_kind}-${i}`} node={p} index={index + i + 1} />
          ))}
        </ol>
      )}
    </li>

  useEffect(() => {
    if (!open) return;
    const onKey = (e: KeyboardEvent) => e.key === "Escape" && onClose();
    document.addEventListener("keydown", onKey);
    return () => document.removeEventListener("keydown", onKey);
  }, [open, onClose]);

  if (!open) return null;

  return (
    <aside
      data-bok-rail
      role="complementary"
      aria-label="Provenance"
      style={{
        position: "fixed",
        insetBlock: 0,
        insetInlineEnd: 0,
        width: "380px",
        overflowY: "auto",
        background: "var(--bok-paper)",
        borderInlineStart: "1px solid var(--bok-neutral-4)",
        padding: "calc(var(--bok-space) * 5)",
        animation: "bok-rail-slide 180ms ease-out both",
      }}
    >
      <button type="button" data-bok-chrome onClick={onClose} aria-label="Close provenance">
        ×
      </button>
      <ol style={{ listStyle: "none", padding: 0 }}>
        <Node node={payload.root} index={0} />
      </ol>
      {payload.inherited_caveats.length > 0 && (
        <section aria-label="Inherited caveats">
          <h3 style={{ fontFamily: "var(--bok-font-display)" }}>Caveats</h3>

            {payload.inherited_caveats.map((c) => (
              <li key={c}>{c}</li>
            ))}
          </ul>
        </section>
      )}
    </aside>
  );
}
```

```ts
import { useCallback, useRef, useState, type KeyboardEvent } from "react";

export interface RovingOptions {
  count: number;
  onActivate?: (index: number) => void;
}

/** One tab stop, arrow/J/K/Home/End movement, Enter to activate.
 *  Shared by FindingsTable and by ledger-sense's review queue -- a queue is not a
 *  table, but the keyboard contract is identical. */
export function useRovingTabIndex({ count, onActivate }: RovingOptions) {
  const [activeIndex, setActiveIndex] = useState(0);
  const refs = useRef<(HTMLElement | null)[]>([]);

  const focusAt = useCallback((next: number) => {
    const clamped = Math.max(0, Math.min(count - 1, next));
    setActiveIndex(clamped);
    refs.current[clamped]?.focus();
  }, [count]);

  const onKeyDown = useCallback(
    (e: KeyboardEvent<HTMLElement>) => {
      const moves: Record<string, number> = {
        ArrowDown: activeIndex + 1, j: activeIndex + 1,
        ArrowUp: activeIndex - 1, k: activeIndex - 1,
        Home: 0, End: count - 1,
      };
      if (e.key in moves) {
        e.preventDefault();
        focusAt(moves[e.key] as number);
        return;
      }
      if (e.key === "Enter") {
        e.preventDefault();
        onActivate?.(activeIndex);
      }
    },
    [activeIndex, count, focusAt, onActivate],
  );

  return {
    activeIndex,
    setActiveIndex: focusAt,
    containerProps: { onKeyDown },
    itemProps: (index: number) => ({
      tabIndex: index === activeIndex ? 0 : -1,
      ref: (el: HTMLElement | null) => {
        refs.current[index] = el;
      },
      onFocus: () => setActiveIndex(index),
    }),
  };
}
```

```ts
import type { ReactNode } from "react";

/** A range in whatever coordinate system this adapter's content uses. */
export interface HighlightRange {
  label?: string;
}

/** The seam that lets one pane serve three content types.
 *  v0.1 ships the text/code adapter. The DOM adapter (node paths) lands in week 13
 *  with selector-drift; the document adapter (page + bbox) in week 5 with
 *  ledger-sense. Designing the seam now is what keeps that a new file rather than
 *  a rewrite of the pane. */
export interface ContentAdapter<TContent, TRange extends HighlightRange> {
  readonly kind: string;
  render(content: TContent, ranges: readonly TRange[]): ReactNode;
}

export interface TextRange extends HighlightRange {
  start: number;
  end: number;
}

export const textAdapter: ContentAdapter<string, TextRange> = {
  kind: "text",
  render(content, ranges) {
    const sorted = [...ranges]
      .map((r) => ({
        ...r,
        start: Math.max(0, Math.min(content.length, r.start)),
        end: Math.max(0, Math.min(content.length, r.end)),
      }))
      .filter((r) => r.end > r.start)
      .sort((a, b) => a.start - b.start);

    const out: ReactNode[] = [];
    let cursor = 0;
    sorted.forEach((r, i) => {
      if (r.start > cursor) out.push(content.slice(cursor, r.start));
      out.push(
        <mark key={i} aria-label={r.label ?? "highlighted evidence"}>
          {content.slice(r.start, r.end)}
        </mark>,
      );
      cursor = r.end;
    });
    if (cursor < content.length) out.push(content.slice(cursor));
    return out;
  },
};
```
